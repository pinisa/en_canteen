function selectTable(id) { selectedTable = id; renderTables(); updateSummary(); } function selectTimeSlot(time, btn) { selectedTime = time; document.querySelectorAll('.time-btn').forEach(b => { b.classList.remove('active', 'bg-red-800', 'text-white', 'border-2', 'shadow-sm'); b.classList.add('bg-slate-50', 'text-slate-700'); }); btn.classList.add('active', 'bg-red-800', 'text-white', 'border-2', 'shadow-sm'); btn.classList.remove('bg-slate-50', 'text-slate-700'); updateSummary(); } function updateSummary() { const tableBadge = document.getElementById('summary-zone-badge'); const tableNo = document.getElementById('summary-table-no'); const timeSlot = document.getElementById('summary-time-slot'); const submitBtnText = document.getElementById('submit-btn-text'); if (tableBadge) tableBadge.innerText = currentFloor === 'upper' ? 'ชั้น 1' : 'ชั้นใต้ดิน'; if (tableNo) tableNo.innerText = `โต๊ะ: ${selectedTable}`; if (timeSlot) timeSlot.innerText = `${selectedTime} น.`; if (submitBtnText) submitBtnText.innerText = `ยืนยันโต๊ะ ${selectedTable} 👈`; } function toggleUserDropdown(event) { event.stopPropagation(); document.getElementById('user-dropdown').classList.toggle('hidden'); } window.addEventListener('click', () => { const dropdown = document.getElementById('user-dropdown'); if (dropdown && !dropdown.classList.contains('hidden')) { dropdown.classList.add('hidden'); } }); function loadUserProfile() { const name = localStorage.getItem('user_name') || 'User'; const email = localStorage.getItem('user_email') || 'user@kkumail.com'; document.getElementById('user-display-name').innerText = name; document.getElementById('dropdown-user-email').innerText = email; } function openChangeNameModal() { document.getElementById('name-modal').classList.remove('hidden'); } function closeChangeNameModal() { document.getElementById('name-modal').classList.add('hidden'); } function saveNewName() { const input = document.getElementById('input-new-name'); if (input.value.trim() !== '') { localStorage.setItem('user_name', input.value.trim()); loadUserProfile(); closeChangeNameModal(); input.value = ''; } } function handleLogout() { localStorage.setItem('is_logged_in', 'false'); window.location.href = 'login.html'; } function confirmBookingAndProceed() { alert(`จองสำเร็จ! โต๊ะ: ${selectedTable} (${currentFloor === 'upper' ? 'ชั้น 1' : 'ชั้นใต้ดิน'}) รอบเวลา: ${selectedTime}`); window.location.href = 'booking.html'; }
